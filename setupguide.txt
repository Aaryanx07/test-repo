# =========================
# PRICE COMPARISON SCRAPER - SETUP GUIDE
# =========================

# Install Node.js
# Recommended Version:
# Node.js v20+ OR v22+
# Download:
# https://nodejs.org/

# =========================
# NPM PACKAGES
# =========================

express
cors
dotenv
playwright
nodemon

# =========================
# INSTALL COMMANDS
# =========================

# Run these commands in terminal:

npm install express cors dotenv playwright nodemon

# Install Playwright browsers
npx playwright install

# =========================
# OPTIONAL BUT RECOMMENDED
# =========================

# Auto restart backend server
npm install --save-dev nodemon

# =========================
# PACKAGE.JSON SCRIPTS
# =========================

# Add this inside package.json

"scripts": {
  "start": "node server.js",
  "dev": "nodemon server.js"
}

# =========================
# RUN PROJECT
# =========================

# Backend
npm run dev

# OR
node server.js

# Frontend
# Use Live Server extension in VS Code
# OR

npx live-server

# =========================
# DEFAULT PORTS
# =========================

# Frontend:
http://127.0.0.1:5500

# Backend:
http://localhost:8000

# =========================

# Recommended:
